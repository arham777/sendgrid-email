def get_recipients_from_csv(filename, column_header):
    """Reads email addresses from the specified column_header of a CSV file."""
    # (Code from previous answer - slightly refined for clarity)
    emails = []
    print(f"\n--- Reading Recipients ---")
    try:
        with open(filename, mode='r', newline='', encoding='utf-8-sig') as csvfile: # Added '-sig' for better BOM handling
            reader = csv.DictReader(csvfile)
            if column_header not in reader.fieldnames:
                print(f"[Error] CSV file '{filename}' missing required header '{column_header}'.")
                print(f"Available headers: {reader.fieldnames}")
                return None
            print(f"Reading emails from column '{column_header}' in '{filename}'...")
            count = 0
            for row_num, row in enumerate(reader, start=2): # start=2 accounts for header row
                if row and column_header in row:
                    email_address = row[column_header].strip()
                    if email_address:
                        emails.append(email_address)
                        count += 1
                    # else: # Optional debug
                    #    print(f"Warning: Empty value in '{column_header}' at row {row_num}")
                # else: # Optional debug
                #    print(f"Warning: Skipping empty/malformed row {row_num}")

            print(f"Found {count} email addresses.")
            if count == 0:
                print(f"[Warning] No emails found in column '{column_header}'.")
            return emails

    except FileNotFoundError:
        print(f"[Error] Recipients CSV file '{filename}' not found.")
        return None
    except Exception as e:
        print(f"[Error] Reading recipients CSV '{filename}': {e}")
        return None

def get_content_from_input():
    """Gets subject, HTML content, and optional image path from user input."""
    print("\n--- Enter Email Content ---")
    subject = input("Enter Subject: ").strip()
    while not subject:
        print("[Error] Subject cannot be empty.")
        subject = input("Enter Subject: ").strip()

    print("Enter HTML Content (press Enter twice to finish):")
    html_lines = []
    try:
        while True:
            line = input()
            if line == "":
                break
            html_lines.append(line)
    except EOFError: # Handles pasting multi-line content ending without double Enter
        pass
    html_content = "\n".join(html_lines).strip()
    while not html_content:
         print("[Error] HTML Content cannot be empty.")
         print("Enter HTML Content (press Enter twice to finish):")
         html_lines = []
         try:
             while True:
                 line = input()
                 if line == "":
                     break
                 html_lines.append(line)
         except EOFError:
             pass
         html_content = "\n".join(html_lines).strip()

    image_path = input("Enter full path to image attachment (optional, press Enter to skip): ").strip()

    return {
        'subject': subject,
        'html_content': html_content,
        'image_path': image_path or None # Store None if empty
    }

def get_content_from_csv(filename, subject_hdr, html_hdr, image_hdr):
    """Reads subject, HTML content, and optional image path from the first data row of a CSV."""
    print(f"\n--- Reading Content from CSV '{filename}' ---")
    try:
        with open(filename, mode='r', newline='', encoding='utf-8-sig') as csvfile:
            reader = csv.DictReader(csvfile)
            required_headers = {subject_hdr, html_hdr}
            available_headers = set(reader.fieldnames)

            if not required_headers.issubset(available_headers):
                missing = required_headers - available_headers
                print(f"[Error] Content CSV '{filename}' is missing required headers: {', '.join(missing)}")
                print(f"Available headers: {reader.fieldnames}")
                return None

            # Read the first data row
            first_row = next(reader, None)
            if not first_row:
                print(f"[Error] Content CSV '{filename}' is empty or contains only headers.")
                return None

            subject = first_row.get(subject_hdr, '').strip()
            html_content = first_row.get(html_hdr, '').strip()
            image_path = None
            if image_hdr in available_headers: # Check if optional image header exists
                 image_path = first_row.get(image_hdr, '').strip() or None

            if not subject:
                print(f"[Error] Subject value is empty in column '{subject_hdr}' in '{filename}'.")
                return None
            if not html_content:
                print(f"[Error] HTML content value is empty in column '{html_hdr}' in '{filename}'.")
                return None

            print(f"Subject: '{subject}'")
            print(f"HTML Content: Read (length: {len(html_content)})")
            print(f"Image Path: {'Provided (' + image_path + ')' if image_path else 'Not Provided'}")

            return {
                'subject': subject,
                'html_content': html_content,
                'image_path': image_path
            }

    except FileNotFoundError:
        print(f"[Error] Content CSV file '{filename}' not found.")
        return None
    except StopIteration:
        print(f"[Error] Content CSV '{filename}' seems to be empty (could not read first row).")
        return None
    except Exception as e:
        print(f"[Error] Reading content CSV '{filename}': {e}")
        return None

def create_attachment(image_path, content_id):
    """Creates a SendGrid Attachment object from an image file path."""
    if not image_path:
        return None

    print(f"\n--- Processing Attachment ---")
    try:
        # Check if file exists before opening
        if not os.path.isfile(image_path):
            print(f"[Error] Image file not found at path: {image_path}")
            return None

        # Read image data
        with open(image_path, 'rb') as f:
            data = f.read()

        # Encode to Base64
        encoded_file = base64.b64encode(data).decode()

        # Guess MIME type
        file_type, _ = mimetypes.guess_type(image_path)
        if not file_type:
            file_type = 'application/octet-stream' # Default if type cannot be guessed
            print(f"[Warning] Could not determine MIME type for '{image_path}'. Using default: {file_type}")

        # Create Attachment object
        attachment = Attachment(
            FileContent(encoded_file),
            FileName(os.path.basename(image_path)), # Use just the filename
            FileType(file_type),
            Disposition('inline'), # 'inline' for embedding, 'attachment' for attaching
            ContentId(content_id)  # Crucial for embedding with <img src='cid:xxx'>
        )
        print(f"Attachment created for '{os.path.basename(image_path)}' (Type: {file_type}, Content-ID: {content_id})")
        return attachment

    except FileNotFoundError:
        # This is redundant due to the isfile check, but good practice
        print(f"[Error] Image file not found during attachment creation: {image_path}")
        return None
    except Exception as e:
        print(f"[Error] Creating attachment for '{image_path}': {e}")
        return None

# --- Main Execution ---
def main():
    """Main function to orchestrate the email sending process."""
    print("=====================================")
    print(" SendGrid Bulk Email Script ")
    print("=====================================")

    if not SENDGRID_API_KEY:
        print("[FATAL] SENDGRID_API_KEY not found in environment variables.")
        print("Please ensure a .env file exists with your key.")
        return # Stop execution

    # 1. Choose Content Source
    content_data = None
    while True:
        mode = input("Get content from (1) Input Fields or (2) CSV File? [1/2]: ").strip()
        if mode == '1':
            content_data = get_content_from_input()
            break
        elif mode == '2':
            content_data = get_content_from_csv(
                CSV_CONTENT_FILENAME,
                CONTENT_SUBJECT_HEADER,
                CONTENT_HTML_HEADER,
                CONTENT_IMAGE_PATH_HEADER
            )
            break
        else:
            print("[Error] Invalid choice. Please enter 1 or 2.")

    if not content_data:
        print("\n[FATAL] Failed to obtain email content. Exiting.")
        return

    # 2. Get Recipients
    recipient_emails = get_recipients_from_csv(CSV_RECIPIENTS_FILENAME, EMAIL_COLUMN_HEADER)

    if not recipient_emails: # Checks for None or empty list
        print("\n[FATAL] Failed to obtain recipients or recipient list is empty. Exiting.")
        return

    # 3. Create Attachment (if image path provided)
    email_attachment = None
    if content_data.get('image_path'):
        email_attachment = create_attachment(content_data['image_path'], ATTACHMENT_CONTENT_ID)
        # Note: If attachment creation fails, we still proceed without it.

    # 4. Build the SendGrid Mail Object
    print("\n--- Building Email ---")
    message = Mail(
        from_email=FROM_EMAIL,
        to_emails=recipient_emails, # SendGrid handles the list efficiently
        subject=content_data['subject'],
        html_content=content_data['html_content']
    )

    if email_attachment:
        message.add_attachment(email_attachment)
        print("Attachment added to the email message.")
    else:
         print("No attachment added.")
         if content_data.get('image_path'):
              print("[Warning] An image path was provided, but attachment creation failed. Sending without image.")


    # 5. Send the Email
    print("\n--- Sending Email via SendGrid ---")
    try:
        sg = SendGridAPIClient(SENDGRID_API_KEY)
        response = sg.send(message)

        print(f"SendGrid Response Status Code: {response.status_code}")
        if 200 <= response.status_code < 300: # Check for 2xx success codes (e.g., 202 Accepted)
            print("[Success] Email request accepted by SendGrid for processing.")
            print(f"Sent to {len(recipient_emails)} recipient(s).")
        else:
            print("[Error] SendGrid returned a non-success status code.")
            print(f"Response Body: {response.body}") # Show error details from SendGrid

    except Exception as e:
        print(f"[FATAL] An error occurred while sending email via SendGrid: {e}")
        if hasattr(e, 'body'):
            print(f"Error details: {e.body}")

    print("\n=====================================")
    print(" Script Finished ")
    print("=====================================")

# --- Run the main function ---
if __name__ == "__main__":
    main()
